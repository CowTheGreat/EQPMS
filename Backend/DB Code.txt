CREATE TABLE users (
    id INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    regno VARCHAR(50) NOT NULL UNIQUE,
    mail VARCHAR(100) NOT NULL UNIQUE,
    pwd VARCHAR(255) NOT NULL,
    board VARCHAR(100),
    password_changed INT DEFAULT 0,
    role VARCHAR(50) NOT NULL DEFAULT 'user',
    employee_type VARCHAR(50) NOT NULL DEFAULT 'Internal'
);

CREATE TABLE qp (
    QP_Ref_no VARCHAR(50) NOT NULL PRIMARY KEY,
    QP_Session VARCHAR(50) NOT NULL,
    QP_Degree VARCHAR(50) NOT NULL,
    QP_Regulation VARCHAR(50) NOT NULL,
    QP_Board VARCHAR(50) NOT NULL,
    QP_Semester INT NOT NULL,
    QP_Subjectcode VARCHAR(50) NOT NULL,
    QP_Subject VARCHAR(255) NOT NULL,
    QP_Faculty_assigned VARCHAR(100) NOT NULL,
    QP_Faculty_assigned_Regno VARCHAR(100) NOT NULL,
    QP_Deadline DATETIME,
    QP_Status VARCHAR(50) DEFAULT 'Assigned',
    QP_Type VARCHAR(50)
);

CREATE TABLE boards (
    id INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    degree VARCHAR(50) NOT NULL,
    regulation VARCHAR(50) NOT NULL,
    board_name VARCHAR(50) NOT NULL,
    subject_code VARCHAR(50) NOT NULL,
    subject_name VARCHAR(255) NOT NULL,
    semester INT NOT NULL DEFAULT 5,
    QP_Type VARCHAR(50) DEFAULT 'T'
);
